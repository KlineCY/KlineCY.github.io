cmake_minimum_required(VERSION 3.16)
project(kline_quant_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(kline_quant cli.cpp)
add_executable(kline_quant_tests test.cpp)
add_executable(kline_quant_advanced_tests test_advanced.cpp)
add_executable(kline_quant_fullstack_tests test_fullstack.cpp)
add_executable(kline_quant_expansion_math_tests test_expansion_math.cpp)
add_executable(kline_quant_expansion_finance_tests test_expansion_finance.cpp)
add_executable(kline_quant_research_math_tests test_research_math.cpp)
add_executable(kline_quant_research_finance_tests test_research_finance.cpp)
add_executable(kline_quant_crossdomain_tests test_crossdomain.cpp)

if(MSVC)
  target_compile_options(kline_quant PRIVATE /W4)
  target_compile_options(kline_quant_tests PRIVATE /W4)
  target_compile_options(kline_quant_advanced_tests PRIVATE /W4)
  target_compile_options(kline_quant_fullstack_tests PRIVATE /W4)
  target_compile_options(kline_quant_expansion_math_tests PRIVATE /W4)
  target_compile_options(kline_quant_expansion_finance_tests PRIVATE /W4)
  target_compile_options(kline_quant_research_math_tests PRIVATE /W4)
  target_compile_options(kline_quant_research_finance_tests PRIVATE /W4)
  target_compile_options(kline_quant_crossdomain_tests PRIVATE /W4)
else()
  target_compile_options(kline_quant PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kline_quant_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kline_quant_advanced_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kline_quant_fullstack_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kline_quant_expansion_math_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(kline_quant_expansion_finance_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(kline_quant_research_math_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(kline_quant_research_finance_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(kline_quant_crossdomain_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

enable_testing()
add_test(NAME kline_quant_numeric_tests COMMAND kline_quant_tests)
add_test(NAME kline_quant_advanced_numeric_tests COMMAND kline_quant_advanced_tests)
add_test(NAME kline_quant_fullstack_numeric_tests COMMAND kline_quant_fullstack_tests)
add_test(NAME kline_quant_expansion_math_tests COMMAND kline_quant_expansion_math_tests)
add_test(NAME kline_quant_expansion_finance_tests COMMAND kline_quant_expansion_finance_tests)
add_test(NAME kline_quant_research_math_tests COMMAND kline_quant_research_math_tests)
add_test(NAME kline_quant_research_finance_tests COMMAND kline_quant_research_finance_tests)
add_test(NAME kline_quant_crossdomain_tests COMMAND kline_quant_crossdomain_tests)
