inline double continuous_discount_factor(double zero_rate, double years) {
    require(std::isfinite(zero_rate) && years >= 0 && std::isfinite(years),
            "Invalid continuously compounded zero rate or maturity");
    return std::exp(-zero_rate * years);
}

inline double continuous_forward_rate(double zero_rate_short, double short_years,
                                       double zero_rate_long, double long_years) {
    require(short_years >= 0 && long_years > short_years &&
            std::isfinite(zero_rate_short) && std::isfinite(zero_rate_long),
            "Forward tenor must be increasing");
    return (zero_rate_long * long_years - zero_rate_short * short_years) /
           (long_years - short_years);
}

inline double term_spread_bps(double long_yield, double short_yield) {
    require(std::isfinite(long_yield) && std::isfinite(short_yield),
            "Yields must be finite");
    return (long_yield - short_yield) * 10000;
}

inline double breakeven_inflation(double nominal_zero_yield, double real_zero_yield) {
    require(std::isfinite(nominal_zero_yield) && std::isfinite(real_zero_yield),
            "Zero rates must be finite");
    // This difference can also include liquidity and inflation risk premia.
    return nominal_zero_yield - real_zero_yield;
}

inline double approximate_parallel_shift_pnl(double bond_market_value,
                                              double modified_duration,
                                              double convexity,
                                              double yield_change) {
    require(bond_market_value >= 0 && modified_duration >= 0 && convexity >= 0 &&
            std::isfinite(bond_market_value) && std::isfinite(yield_change),
            "Invalid rate-shock parameters");
    double fractional_change =
        -modified_duration * yield_change +
        0.5 * convexity * yield_change * yield_change;
    return bond_market_value * fractional_change;
}

inline double duration_hedge_notional(double asset_market_value,
                                       double asset_duration, double hedge_duration) {
    require(asset_market_value >= 0 && asset_duration >= 0 && hedge_duration > 0,
            "Invalid asset or hedge duration");
    // Sign convention: negative means short the hedging instrument.
    return -asset_market_value * asset_duration / hedge_duration;
}

inline double fx_forward_quote_per_base(double spot_quote_per_base,
                                         double base_annual_rate,
                                         double quote_annual_rate,
                                         double years) {
    require(spot_quote_per_base > 0 && years >= 0 &&
            std::isfinite(base_annual_rate) && std::isfinite(quote_annual_rate),
            "Invalid FX forward inputs");
    require(1 + base_annual_rate * years > 0 &&
            1 + quote_annual_rate * years > 0, "Money-market factors must be positive");
    // Spot is units of quote currency for one unit of base currency.
    // Covered interest parity: forward = spot*(1+r_quote*T)/(1+r_base*T).
    return spot_quote_per_base * (1 + quote_annual_rate * years) /
           (1 + base_annual_rate * years);
}

inline double futures_roll_yield(double expiring_contract_price,
                                  double deferred_contract_price) {
    require(expiring_contract_price > 0 && deferred_contract_price > 0 &&
            std::isfinite(expiring_contract_price) && std::isfinite(deferred_contract_price),
            "Futures prices must be positive for this proportional roll measure");
    // A simplified static roll measure; realized futures returns also depend on
    // subsequent changes in the contracts' prices.
    return expiring_contract_price / deferred_contract_price - 1;
}

inline double excess_bond_return_approx(double carry_yield,
                                         double modified_duration,
                                         double yield_change, double convexity) {
    require(modified_duration >= 0 && convexity >= 0 &&
            std::isfinite(carry_yield) && std::isfinite(yield_change),
            "Invalid approximate bond return inputs");
    return carry_yield - modified_duration * yield_change +
           0.5 * convexity * yield_change * yield_change;
}

#include "curve_models.hpp"
#include <cmath>
#include <vector>
namespace kline {
inline double nelson_siegel_zero(double maturity,double level,
    double slope,double curvature,double decay){
    require(std::isfinite(maturity)&&maturity>=0&&
            std::isfinite(level)&&std::isfinite(slope)&&
            std::isfinite(curvature)&&std::isfinite(decay)&&decay>0,
            "Invalid Nelson-Siegel zero inputs");
    if(maturity==0)return level+slope;
    double x=maturity/decay;
    double loading=-std::expm1(-x)/x;
    return level+slope*loading+
           curvature*(loading-std::exp(-x));
}
inline ZeroCurve nelson_siegel_curve(const std::vector<double>& times,
    double level,double slope,double curvature,double decay){
    check_finite(times);
    std::vector<double> rates;
    for(int i=0;i<int(times.size());i++){
        require(times[i]>0&&(i==0||times[i]>times[i-1]),
                "Nelson Siegel tenor order");
        rates.push_back(nelson_siegel_zero(times[i],level,slope,
                                           curvature,decay));
    }
    return {times,rates};
}

#pragma once
#include "fixed_income_advanced.hpp"
#include <cmath>
#include <vector>
namespace kline {
struct ContinuousBondRisk{
    double price=0,duration=0,convexity=0,dv01=0;
};
inline ContinuousBondRisk continuous_bond_risk(
    const std::vector<double>& times,const std::vector<double>& cashflows,
    double yield){
    check_finite(times);check_finite(cashflows);
    require(times.size()==cashflows.size()&&
            std::isfinite(yield),"Invalid bond risk cashflows");
    ContinuousBondRisk r;
    double first=0,second=0;
    for(int i=0;i<int(times.size());i++){
        require(times[i]>0,"Bond times must be future");
        double pv=cashflows[i]*std::exp(-yield*times[i]);
        r.price+=pv;first+=times[i]*pv;second+=times[i]*times[i]*pv;
    }
    require(std::isfinite(r.price)&&r.price>0&&
            std::isfinite(first)&&std::isfinite(second),
            "Bond price, duration or convexity overflow");
    r.duration=first/r.price;
    r.convexity=second/r.price;
    r.dv01=first*1e-4;
    return r;
}
inline double bond_shock_pnl_approx(double price,double duration,
                                   double convexity,double rate_change){
    require(std::isfinite(price)&&price>0&&std::isfinite(duration)&&
            std::isfinite(convexity)&&std::isfinite(rate_change),
            "Invalid rate shock parameters");
    return price*(-duration*rate_change+
                  0.5*convexity*rate_change*rate_change);
}

#pragma once
#include "stats.hpp"
#include <cmath>
#include <vector>
namespace kline {
inline double net_present_value(double initial_investment,
    const std::vector<double>& future_cashflows,double discount){
    check_finite(future_cashflows);
    require(std::isfinite(initial_investment)&&
            std::isfinite(discount)&&discount>-1,
            "Invalid NPV inputs");
    double value=-initial_investment;
    for(int t=0;t<int(future_cashflows.size());t++)
        value+=future_cashflows[t]/std::pow(1+discount,t+1);
    return value;
}
inline double weighted_average_cost_of_capital(double equity_weight,
    double cost_equity,double pretax_cost_debt,double tax_rate){
    require(std::isfinite(equity_weight)&&equity_weight>=0&&equity_weight<=1&&
            std::isfinite(cost_equity)&&std::isfinite(pretax_cost_debt)&&
            std::isfinite(tax_rate)&&tax_rate>=0&&tax_rate<=1,
            "Invalid WACC values");
    return equity_weight*cost_equity+
           (1-equity_weight)*pretax_cost_debt*(1-tax_rate);
}
inline double perpetuity_value(double next_cashflow,double growth,
                                double discount){
    require(std::isfinite(next_cashflow)&&std::isfinite(growth)&&
            std::isfinite(discount)&&discount>growth,
            "Perpetuity growth must be below discount rate");
    return next_cashflow/(discount-growth);
}

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
#include <numeric>
#include <vector>
namespace kline {
inline double liquidation_days(const std::vector<double>& shares,
    const std::vector<double>& average_daily_volume,double participation){
    check_finite(shares);check_finite(average_daily_volume);
    require(shares.size()==average_daily_volume.size()&&
            std::isfinite(participation)&&participation>0&&participation<=1,
            "Invalid liquidation inputs");
    double days=0;
    for(int i=0;i<int(shares.size());i++){
        require(average_daily_volume[i]>0,"Nonpositive ADV");
        days=std::max(days,std::fabs(shares[i])/
             (participation*average_daily_volume[i]));
    }
    return days;
}
inline double staged_liquidation_cost(const std::vector<double>& quantities,
    const std::vector<double>& spreads,
    const std::vector<double>& participation_rates,
    double impact_coefficient){
    check_finite(quantities);check_finite(spreads);
    check_finite(participation_rates);
    require(quantities.size()==spreads.size()&&
            quantities.size()==participation_rates.size()&&
            std::isfinite(impact_coefficient)&&impact_coefficient>=0,
            "Invalid liquidity cost arrays");
    double total=0;
    for(int i=0;i<int(quantities.size());i++){
        require(spreads[i]>=0&&participation_rates[i]>=0,
                "Negative liquidity cost rate");
        total+=std::fabs(quantities[i])*(spreads[i]+
                 impact_coefficient*std::sqrt(participation_rates[i]));
    }
    return total;
}
