inline std::vector<DatedValue> aligned_intersection(
    const std::vector<DatedValue>& a,const std::vector<DatedValue>& b,
    std::vector<double>& paired_b){
    require(!a.empty()&&!b.empty(),"Dated series required");
    auto validate=[](const std::vector<DatedValue>& values){
        for(int i=0;i<int(values.size());i++){
            require(std::isfinite(values[i].value),"Nonfinite dated value");
            if(i>0)require(values[i].date>values[i-1].date,
                            "Dates must be strictly ordered");
        }
    };

    int i=0,j=0;
    std::vector<DatedValue> out;paired_b.clear();
    while(i<int(a.size())&&j<int(b.size())){

            out.push_back(a[i]);paired_b.push_back(b[j].value);i++;j++;
        }else if(a[i].date<b[j].date)i++;else j++;
    }
    require(!out.empty(),"No overlapping market dates");
    return out;
}

namespace kline {
struct LabeledPeriod {
    int start=0;
    int end=0;
};

inline std::vector<int> purge_label_intervals(
    const std::vector<LabeledPeriod>& training,
    const LabeledPeriod& test,int embargo=0) {
    require(test.end>test.start&&embargo>=0,
            "Invalid validation interval or embargo");
    std::vector<int> keep;
    for(int i=0;i<int(training.size());i++){
        auto event=training[i];
        require(event.end>event.start,"Invalid training event interval");
        bool overlaps=event.start<test.end&&event.end>test.start;
        bool embargoed=event.start>=test.end&&
                        event.start<test.end+embargo;
        if(!overlaps&&!embargoed)keep.push_back(i);
    }
    return keep;
}

inline int eligible_training_count(const std::vector<int>& maturity,
                                    int decision,int gap=0) {
    require(!maturity.empty()&&decision>=0&&gap>=0&&decision>=gap,
            "Invalid label maturity gate");
    int cutoff=decision-gap,count=0;
    for(int i=0;i<int(maturity.size());i++){
        require(maturity[i]>i,"Label maturity must follow feature time");
        if(i<cutoff&&maturity[i]<=cutoff)count++;
    }
    return count;
}

inline std::vector<int> chronological_holdout_indices(
    int observations,int last_test_count,int gap=0) {
    require(observations>=3&&last_test_count>0&&gap>=0&&
            last_test_count+gap<observations,
            "Invalid chronological holdout parameters");
    std::vector<int> labels;
    for(int t=observations-last_test_count;t<observations;t++)
        labels.push_back(t);
    return labels;
}

#include <cmath>

namespace kline {
struct BarrierLabel{
    int start=0,end=0,direction=0;
    double realized_return=0;
};
inline BarrierLabel label_triple_barrier(const std::vector<double>& prices,
    int start,int horizon,double profit_take,double stop_loss){
    check_finite(prices);
    require(start>=0&&start<int(prices.size())-1&&
            horizon>=1&&std::isfinite(profit_take)&&profit_take>0&&
            std::isfinite(stop_loss)&&stop_loss>0,
            "Invalid triple barrier parameters");
    for(double p:prices)require(p>0,"Triple barrier needs positive prices");
    int last=std::min(int(prices.size())-1,start+horizon);
    double entry=prices[start];BarrierLabel result;
    result.start=start;result.end=last;
    for(int t=start+1;t<=last;t++){
        double r=prices[t]/entry-1;
        if(r>=profit_take){result.end=t;result.direction=1;break;}
        if(r<=-stop_loss){result.end=t;result.direction=-1;break;}
    }
    result.realized_return=prices[result.end]/entry-1;
    return result;
}
inline std::vector<BarrierLabel> label_all_barriers(
    const std::vector<double>& prices,int horizon,
    double profit_take,double stop_loss){
    check_finite(prices);
    std::vector<BarrierLabel> out;
    require(horizon>=1&&horizon<int(prices.size()),
            "Full triple-barrier horizon exceeds sample");
    for(int t=0;t+horizon<int(prices.size());t++)
        out.push_back(label_triple_barrier(prices,t,horizon,
                                           profit_take,stop_loss));
    return out;
}

#include "stats.hpp"
#include <cmath>
#include <vector>
namespace kline {
struct TStatistic{
    double estimate=0,standard_error=0,t_value=0;
};
inline TStatistic welch_t_statistic(const std::vector<double>& a,
                                    const std::vector<double>& b){
    check_finite(a);check_finite(b);
    require(a.size()>=2&&b.size()>=2,"Welch test needs two observations per group");
    double va=variance(a)/a.size(),vb=variance(b)/b.size();
    double se=std::sqrt(va+vb);
    require(se>0,"Welch statistic undefined for zero variance");
    return {mean(a)-mean(b),se,(mean(a)-mean(b))/se};
}
inline TStatistic paired_t_statistic(const std::vector<double>& a,
                                     const std::vector<double>& b){
    check_finite(a);check_finite(b);
    require(a.size()==b.size()&&a.size()>=2,"Paired test shape mismatch");
    std::vector<double> d(a.size());
    for(int i=0;i<int(a.size());i++)d[i]=a[i]-b[i];
    double se=std::sqrt(variance(d)/d.size());
    require(se>0,"Paired t undefined for zero differences variance");
    return {mean(d),se,mean(d)/se};
}
inline double population_standard_error(const std::vector<double>& data){
    check_finite(data);
    require(data.size()>=2,"Standard error needs two observations");
    return std::sqrt(variance(data)/data.size());
}

#pragma once
#include "learning.hpp"
#include <vector>
namespace kline {
struct OriginSplit{
    int train_start=0,train_end=0,test_start=0,test_end=0;
};
inline std::vector<OriginSplit> rolling_origin_splits(
    int observations,int initial_train,int test_length,
    int step,int gap=0,int max_train=0){
    require(initial_train>=3&&test_length>=1&&step>=1&&gap>=0&&
            observations>=initial_train+gap+test_length&&
            max_train>=0,"Invalid rolling origin parameters");
    std::vector<OriginSplit> splits;
    for(int end=initial_train;end+gap+test_length<=observations;end+=step){
        int start=max_train?std::max(0,end-max_train):0;
        require(end-start>=3,"Short training segment");
        splits.push_back({start,end,end+gap,end+gap+test_length});
    }
    return splits;
}
inline std::vector<double> rolling_origin_ridge_errors(
    const Matrix& x,const std::vector<double>& y,
    const std::vector<OriginSplit>& splits,double penalty){
    check_feature_table(x,y);
    require(penalty>0&&!splits.empty(),"Invalid rolling origin Ridge penalty");
    std::vector<double> errors;
    for(const auto& segment:splits){
        require(segment.train_start>=0&&segment.train_end<=segment.test_start&&
                segment.test_start<segment.test_end&&
                segment.test_end<=int(x.size()),
                "Rolling origin split leaks future data");
        Matrix training(x.begin()+segment.train_start,
                        x.begin()+segment.train_end);
        std::vector<double> labels(y.begin()+segment.train_start,
                                   y.begin()+segment.train_end);
        auto fit=fit_ridge(training,labels,penalty);
        for(int i=segment.test_start;i<segment.test_end;i++)
            errors.push_back(y[i]-ridge_predict(fit,x[i]));
    }
    return errors;
}

#include <cmath>
#include <vector>
namespace kline {
struct ExecutionAttribution{
    double spread_cost=0,market_movement=0,fees=0,total=0;
};
inline ExecutionAttribution execution_cost_attribution(
    const std::vector<Fill>& fills,
    const std::vector<double>& contemporaneous_midpoints,
    double arrival_mid){
    require(fills.size()==contemporaneous_midpoints.size()&&
            std::isfinite(arrival_mid)&&arrival_mid>0,
            "Invalid execution benchmark alignment");
    check_finite(contemporaneous_midpoints);
    ExecutionAttribution result;
    for(int i=0;i<int(fills.size());i++){
        require(std::isfinite(fills[i].signed_shares)&&
                std::isfinite(fills[i].price)&&fills[i].price>0&&
                std::isfinite(fills[i].commission)&&fills[i].commission>=0&&
                contemporaneous_midpoints[i]>0,
                "Invalid execution fill or midpoint");
        result.spread_cost+=fills[i].signed_shares*
                            (fills[i].price-contemporaneous_midpoints[i]);
        result.market_movement+=fills[i].signed_shares*
                               (contemporaneous_midpoints[i]-arrival_mid);
        result.fees+=fills[i].commission;
    }
    result.total=result.spread_cost+result.market_movement+result.fees;
    return result;
}
