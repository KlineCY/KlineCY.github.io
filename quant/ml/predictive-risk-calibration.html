namespace kline {
struct PoissonFit{std::vector<double> coefficients;};
inline PoissonFit fit_poisson_glm_sgd(
    const Matrix& features,const std::vector<double>& counts,
    double learning_rate,int epochs,double l2=0.001){
    require(features.size()==counts.size()&&!features.empty()&&
            std::isfinite(learning_rate)&&learning_rate>0&&
            epochs>=1&&epochs<=100000&&std::isfinite(l2)&&l2>=0,
            "Invalid Poisson GLM optimizer inputs");
    check_finite(counts);
    int p=int(features.front().size());
    require(p>0,"Poisson GLM requires at least one feature");
    for(const auto& row:features){
        check_finite(row);
        require(int(row.size())==p,"Poisson regression feature shape");
    }
    double average=mean(counts);
    for(double c:counts)require(c>=0&&std::floor(c)==c,
                                "Poisson target requires count outcomes");
    std::vector<double> beta(p+1,0);
    beta[0]=std::log(std::max(0.001,average));
    for(int epoch=0;epoch<epochs;epoch++){
        std::vector<double> gradient(p+1,0);
        for(int t=0;t<int(counts.size());t++){
            double linear=beta[0];
            for(int j=0;j<p;j++)linear+=beta[j+1]*features[t][j];
            double expected=std::exp(std::clamp(linear,-20.0,20.0));
            double error=std::clamp(expected-counts[t],-100.0,100.0)/
                         counts.size();
            gradient[0]+=error;
            for(int j=0;j<p;j++)gradient[j+1]+=error*features[t][j];
        }
        double eta=learning_rate/std::sqrt(epoch+1.0);
        for(int j=0;j<p+1;j++)
            beta[j]-=eta*(gradient[j]+(j?l2*beta[j]:0));
    }
    return {beta};
}
inline double predict_poisson_glm(const PoissonFit& model,
                                  const std::vector<double>& feature){
    check_finite(feature);
    require(feature.size()+1==model.coefficients.size(),
            "Poisson prediction dimensionality mismatch");
    double score=model.coefficients[0];
    for(int j=0;j<int(feature.size());j++)
        score+=feature[j]*model.coefficients[j+1];
    return std::exp(std::clamp(score,-20.0,20.0));
}

#include "portfolio.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
inline double multiclass_negative_log_likelihood(
    const Matrix& probabilities,const std::vector<int>& labels,
    double probability_floor=1e-12){
    require(!probabilities.empty()&&probabilities.size()==labels.size()&&
            std::isfinite(probability_floor)&&probability_floor>0&&
            probability_floor<1,"Invalid multiclass log loss shape");
    int classes=int(probabilities.front().size());
    require(classes>=2,"Multiclass log loss needs at least two classes");
    double loss=0;
    for(int i=0;i<int(labels.size());i++){
        check_finite(probabilities[i]);
        require(int(probabilities[i].size())==classes&&
                labels[i]>=0&&labels[i]<classes,
                "Multiclass prediction dimension or label");
        double total=0;
        for(double p:probabilities[i]){
            require(p>=0&&p<=1,"Invalid multiclass probability");
            total+=p;
        }
        require(std::fabs(total-1)<1e-8,"Class probabilities must sum to one");
        loss-=std::log(std::max(probability_floor,
                               probabilities[i][labels[i]]))/labels.size();
    }
    return loss;
}
} // namespace kline

#include "stats.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
struct ReliabilityBin{int count=0;double mean_forecast=0,event_rate=0;};
inline std::vector<ReliabilityBin> binary_reliability_bins(
    const std::vector<double>& probabilities,const std::vector<int>& labels,
    int bins){
    check_finite(probabilities);
    require(probabilities.size()==labels.size()&&bins>=2&&bins<=1000,
            "Invalid probability calibration bins");
    std::vector<ReliabilityBin> result(bins);
    for(int i=0;i<int(labels.size());i++){
        double p=probabilities[i];
        require(p>=0&&p<=1&&(labels[i]==0||labels[i]==1),
                "Invalid probability or binary outcome");
        int bin=std::min(bins-1,int(p*bins));
        result[bin].count++;
        result[bin].mean_forecast+=p;
        result[bin].event_rate+=labels[i];
    }
    for(auto& bucket:result)if(bucket.count>0){
        bucket.mean_forecast/=bucket.count;
        bucket.event_rate/=bucket.count;
    }
    return result;
}
} // namespace kline

namespace kline {
struct CoverageSummary{double empirical_coverage=0,average_width=0;
    int observations=0;};
inline CoverageSummary interval_coverage_report(
    const std::vector<double>& actual,
    const std::vector<double>& lower,const std::vector<double>& upper){
    check_finite(actual);check_finite(lower);check_finite(upper);
    require(actual.size()==lower.size()&&actual.size()==upper.size(),
            "Prediction intervals must align to actuals");
    int covered=0;double width=0;
    for(int i=0;i<int(actual.size());i++){
        require(lower[i]<=upper[i],"Lower interval exceeds upper interval");
        covered+=actual[i]>=lower[i]&&actual[i]<=upper[i];
        width+=upper[i]-lower[i];
    }
    return {double(covered)/actual.size(),width/actual.size(),
            int(actual.size())};
}

5．Welford在线均值与方差

5.1　对象、目标和定义

实时市场流不断产生新观测，在线统计通过维护样本数量、均值与二阶中心矩，就可以不保存完整历史而逐笔更新基础描述性量度。

$$\mu_n=\mu_{n-1}+\frac{x_n-\mu_{n-1}}n$$

C++17算法源码：online_moments_state.hpp　验证输入与预期数值

5.2　模型推导及基本公式

#include "stats.hpp"
#include <cmath>
namespace kline {
struct OnlineMomentsState{long long observations=0;double mean_value=0,m2=0;};
inline OnlineMomentsState online_moment_update(OnlineMomentsState state,double value){
    require(std::isfinite(value)&&state.observations>=0&&
            state.observations<1000000000000LL,"Invalid online moment state");
    state.observations++;
    double delta=value-state.mean_value;
    state.mean_value+=delta/state.observations;
    state.m2+=delta*(value-state.mean_value);
    return state;
}
inline double online_moment_sample_variance(const OnlineMomentsState& state){
    require(state.observations>=2&&state.m2>=-1e-9,
            "Sample online variance needs two observations");
    return std::max(0.0,state.m2/(state.observations-1));
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
struct PredictionMetrics{double mae=0,rmse=0,mean_bias=0;};
inline PredictionMetrics out_of_sample_regression_metrics(
    const std::vector<double>& actual,const std::vector<double>& forecast){
    check_finite(actual);check_finite(forecast);
    require(actual.size()==forecast.size(),
            "Regression forecasts and actuals must align");
    double abs_sum=0,square=0,bias=0;
    for(int i=0;i<int(actual.size());i++){
        double error=forecast[i]-actual[i];
        abs_sum+=std::fabs(error);square+=error*error;bias+=error;
    }
    double n=double(actual.size());
    return {abs_sum/n,std::sqrt(square/n),bias/n};
}
} // namespace kline
