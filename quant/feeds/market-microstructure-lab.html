inline double optimal_probability_threshold(
    double gain_if_correct,double loss_if_incorrect,double trading_cost){

    require(std::isfinite(probability)&&probability>=0&&probability<=1&&

            std::isfinite(trading_cost)&&trading_cost>=0,

#include "orderbook_replay.hpp"

namespace kline {
struct MarketSweep{double filled=0,remaining=0,vwap=0,notional=0;};
inline MarketSweep sweep_visible_orderbook(
    const BookReplay& book,bool buy,double desired_quantity){
    require(std::isfinite(desired_quantity)&&desired_quantity>=0,
            "Invalid market order quantity");
    validate_uncrossed_book(book);
    MarketSweep result;result.remaining=desired_quantity;
    if(buy){
        for(const auto& [price,size]:book.asks){
            double filled=std::min(size,result.remaining);
            result.notional+=filled*price;result.filled+=filled;
            result.remaining-=filled;
            if(result.remaining<=1e-12)break;
        }
    }else{
        for(auto it=book.bids.rbegin();it!=book.bids.rend();++it){
            double filled=std::min(it->second,result.remaining);
            result.notional+=filled*it->first;result.filled+=filled;
            result.remaining-=filled;
            if(result.remaining<=1e-12)break;
        }
    }
    if(result.filled>0)result.vwap=result.notional/result.filled;
    return result;
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
inline double market_volume_weighted_price(
    const std::vector<double>& prices,const std::vector<double>& volumes){
    check_finite(prices);check_finite(volumes);
    require(prices.size()==volumes.size(),"VWAP prices and volumes misaligned");
    double total_vol=0,total_notional=0;
    for(int i=0;i<int(prices.size());i++){
        require(prices[i]>0&&volumes[i]>=0,"Invalid VWAP trade");
        total_vol+=volumes[i];
        total_notional+=prices[i]*volumes[i];
    }
    require(total_vol>0,"VWAP needs nonzero traded volume");
    return total_notional/total_vol;
}
inline double vwap_slippage_fraction(
    double execution_price,double benchmark,bool buy){
    require(std::isfinite(execution_price)&&execution_price>0&&
            std::isfinite(benchmark)&&benchmark>0,
            "Invalid VWAP execution benchmark");
    return (buy?1:-1)*(execution_price-benchmark)/benchmark;
}
} // namespace kline

namespace kline {
struct ArrivalDurationStats{double average_seconds=0,coefficient_of_variation=0;};
inline ArrivalDurationStats trade_arrival_duration_stats(
    const std::vector<double>& seconds){
    check_finite(seconds);
    require(seconds.size()>=4,"Trade duration statistics need four timestamps");
    std::vector<double> durations;
    for(int i=1;i<int(seconds.size());i++){
        require(seconds[i]>seconds[i-1],"Trade timestamps must increase");
        durations.push_back(seconds[i]-seconds[i-1]);
    }
    double average=mean(durations);
    return {average,std::sqrt(variance(durations,false))/average};
}

    double volume=0,notional=0;
    for(int i=0;i<int(trade_prices.size());i++){
        require(trade_prices[i]>0&&trade_sizes[i]>=0,
                "Invalid trade for volume bar");
        double remain=trade_sizes[i];
        while(remain>1e-12){
            double part=std::min(remain,target_volume-volume);
            volume+=part;notional+=trade_prices[i]*part;remain-=part;
            if(volume>=target_volume-1e-12){
                bars.push_back({target_volume,notional/volume});
                volume=0;notional=0;
            }
        }
    }
    if(volume>1e-12)bars.push_back({volume,notional/volume});
    return bars;
}
} // namespace kline

namespace kline {
inline double inventory_variance_penalty(
    double stock_units,double volatility_fraction,double mid_price,
    double horizon_years,double risk_coefficient){
    require(std::isfinite(stock_units)&&
            std::isfinite(volatility_fraction)&&volatility_fraction>=0&&
            std::isfinite(mid_price)&&mid_price>0&&
            std::isfinite(horizon_years)&&horizon_years>=0&&
            std::isfinite(risk_coefficient)&&risk_coefficient>=0,
            "Invalid inventory uncertainty inputs");
    double money_risk=stock_units*mid_price*volatility_fraction;
    return 0.5*risk_coefficient*money_risk*money_risk*horizon_years;
}
inline double inventory_mark_to_market_pnl(
    double stock_units,double prior_mid,double current_mid){
    require(std::isfinite(stock_units)&&
            std::isfinite(prior_mid)&&prior_mid>0&&
            std::isfinite(current_mid)&&current_mid>0,
            "Invalid inventory mark to market");
    return stock_units*(current_mid-prior_mid);
}
