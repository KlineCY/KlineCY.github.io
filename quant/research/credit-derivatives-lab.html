namespace kline {
inline double maximum_call_vertical_violation(
   const std::vector<double>& strikes,const std::vector<double>& call_prices){
    check_finite(strikes);check_finite(call_prices);
    require(strikes.size()==call_prices.size()&&strikes.size()>=2,
            "Invalid call price strike grid");
    double violation=0;
    for(int i=1;i<int(strikes.size());i++){
        require(strikes[i]>strikes[i-1]&&call_prices[i]>=0&&
                call_prices[i-1]>=0,
                "Non-increasing strike or negative call price");
        violation=std::max(violation,call_prices[i]-call_prices[i-1]);
    }
    return violation;
}
inline double maximum_call_convexity_violation(
   const std::vector<double>& strikes,const std::vector<double>& prices){
    check_finite(strikes);check_finite(prices);
    require(strikes.size()==prices.size()&&strikes.size()>=3,
            "Invalid convexity strike grid");
    double violation=0;
    for(int i=1;i+1<int(strikes.size());i++){
        require(strikes[i-1]<strikes[i]&&strikes[i]<strikes[i+1],
                "Call strikes must be strictly ordered");
        double slope_left=(prices[i]-prices[i-1])/
                          (strikes[i]-strikes[i-1]);
        double slope_right=(prices[i+1]-prices[i])/
                           (strikes[i+1]-strikes[i]);
        violation=std::max(violation,slope_left-slope_right);
    }
    return violation;
}

inline double dupire_local_volatility_squared(
    double strike,double call_price,double time_derivative,
    double strike_derivative,double strike_second_derivative,
    double rate,double dividend_yield){
    require(std::isfinite(strike)&&strike>0&&std::isfinite(call_price)&&
            call_price>=0&&std::isfinite(time_derivative)&&
            std::isfinite(strike_derivative)&&
            std::isfinite(strike_second_derivative)&&
            strike_second_derivative>0&&std::isfinite(rate)&&
            std::isfinite(dividend_yield),"Invalid Dupire local-vol inputs");
    double numerator=time_derivative+(rate-dividend_yield)*
                     strike*strike_derivative+dividend_yield*call_price;
    double denominator=0.5*strike*strike*strike_second_derivative;
    require(numerator>=0,"Negative Dupire local variance numerator");
    return numerator/denominator;
}

#include "credit_survival_curve.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
struct CDSLegs{double premium_annuity=0,default_protection=0,fair_spread=0;};
inline CDSLegs cds_discrete_legs(
    const HazardCurve& curve,double maturity,double rate,
    double recovery,int intervals_per_year=4){
    require(std::isfinite(maturity)&&maturity>0&&std::isfinite(rate)&&
            std::isfinite(recovery)&&recovery>=0&&recovery<1&&
            intervals_per_year>=1&&intervals_per_year<=365,
            "Invalid CDS protection contract");
    int n=int(std::ceil(maturity*intervals_per_year));
    require(n>0&&n<=200000,"Excessive CDS time grid");
    double annuity=0,protection=0,prior=1,previous_time=0;
    for(int k=1;k<=n;k++){
        double t=std::min(maturity,k/double(intervals_per_year));
        double delta=t-previous_time;
        double s=survival_probability(curve,t);
        annuity+=delta*std::exp(-rate*t)*s;
        protection+=(prior-s)*std::exp(-rate*(previous_time+t)/2)*(1-recovery);
        prior=s;previous_time=t;
    }
    require(annuity>0,"Nonpositive CDS premium annuity");
    return {annuity,protection,protection/annuity};
}
} // namespace kline

    maintenance_shortfall=0;};
inline std::vector<FuturesSettlementStep> futures_daily_settlement(
    const std::vector<double>& settlement_prices,double contracts,
    double multiplier,double cash_collateral,double maintenance_requirement){
    check_finite(settlement_prices);
    require(settlement_prices.size()>=2&&
            std::isfinite(contracts)&&std::isfinite(multiplier)&&multiplier>0&&
            std::isfinite(cash_collateral)&&cash_collateral>=0&&
            std::isfinite(maintenance_requirement)&&maintenance_requirement>=0,
            "Invalid futures variation margin inputs");
    double equity=cash_collateral;
    std::vector<FuturesSettlementStep> report;
    for(int i=1;i<int(settlement_prices.size());i++){
        double variation=contracts*multiplier*
                         (settlement_prices[i]-settlement_prices[i-1]);
        equity+=variation;
        report.push_back({variation,equity,
                          std::max(0.0,maintenance_requirement-equity)});
    }
    return report;
}
} // namespace kline
