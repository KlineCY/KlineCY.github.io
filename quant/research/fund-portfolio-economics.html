#include "stats.hpp"
#include <cmath>
#include <vector>
namespace kline {
inline double period_time_weighted_return(
    const std::vector<double>& start_values,
    const std::vector<double>& end_values_before_external_flow){
    check_finite(start_values);check_finite(end_values_before_external_flow);
    require(start_values.size()==end_values_before_external_flow.size(),
            "Time weighted subperiod dimensions differ");
    double compound=1;
    for(int i=0;i<int(start_values.size());i++){
        require(start_values[i]>0&&end_values_before_external_flow[i]>=0,
                "Invalid time weighted portfolio values");
        compound*=end_values_before_external_flow[i]/start_values[i];
    }
    return compound-1;
}
inline double dietz_modified_return(
    double beginning_equity,double ending_equity,
    const std::vector<double>& external_contributions,
    const std::vector<double>& remaining_time_fraction){
    check_finite(external_contributions);check_finite(remaining_time_fraction);
    require(external_contributions.size()==remaining_time_fraction.size()&&
            std::isfinite(beginning_equity)&&beginning_equity>0&&
            std::isfinite(ending_equity)&&ending_equity>=0,
            "Invalid Modified Dietz cashflow report");
    double contribution=0,weighted=beginning_equity;
    for(int i=0;i<int(external_contributions.size());i++){
        require(remaining_time_fraction[i]>=0&&remaining_time_fraction[i]<=1,
                "Contribution weights must be inside period");
        contribution+=external_contributions[i];
        weighted+=external_contributions[i]*remaining_time_fraction[i];
    }
    require(weighted>0,"Modified Dietz invested capital nonpositive");
    return (ending_equity-beginning_equity-contribution)/weighted;
}
} // namespace kline

inline CapacityEconomics square_root_capacity_profit(
    double deployed_capital,double gross_alpha_fraction,
    double annual_turnover,double market_impact_coefficient){
    require(std::isfinite(deployed_capital)&&deployed_capital>=0&&
            std::isfinite(gross_alpha_fraction)&&
            std::isfinite(annual_turnover)&&annual_turnover>=0&&
            std::isfinite(market_impact_coefficient)&&
            market_impact_coefficient>=0,"Invalid strategy capacity inputs");

    double impact=market_impact_coefficient*annual_turnover*
                  std::pow(deployed_capital,1.5);
    return {gross,impact,gross-impact};
}
inline double square_root_capacity_optimal_capital(
    double alpha_fraction,double turnover,double impact_coefficient){
    require(std::isfinite(alpha_fraction)&&alpha_fraction>=0&&
            std::isfinite(turnover)&&turnover>0&&
            std::isfinite(impact_coefficient)&&impact_coefficient>0,
            "Capacity optimum needs positive cost and turnover");

    return ratio*ratio;
}

inline Matrix shrink_covariance_to_diagonal(const Matrix& raw,double intensity){
    check_covariance(raw);
    require(std::isfinite(intensity)&&intensity>=0&&intensity<=1,
            "Covariance shrinkage must lie in [0,1]");
    Matrix result=raw;
    for(int i=0;i<int(raw.size());i++)
        for(int j=0;j<int(raw.size());j++)
            if(i!=j)result[i][j]*=1-intensity;
    return result;
}
inline double covariance_average_variance(const Matrix& raw){
    check_covariance(raw);

#include "portfolio.hpp"
#include <cmath>
#include <numeric>
namespace kline {
inline std::vector<double> positive_covariance_risk_parity(
    const Matrix& covariance,int iterations=1000,double tolerance=1e-9){
    check_covariance(covariance);
    int n=int(covariance.size());
    require(iterations>=1&&tolerance>0&&
            std::isfinite(tolerance),"Invalid risk parity iterations");
    for(int i=0;i<n;i++)for(int j=0;j<n;j++){
        require(covariance[i][j]>=0,
                "Multiplicative risk parity requires nonnegative covariance");
        if(i==j)require(covariance[i][i]>0,
                        "Risk parity needs positive asset variance");
    }
    std::vector<double> w(n,1.0/n);
    for(int step=0;step<iterations;step++){
        auto marginal=covariance_times_weights(covariance,w);
        double total=0;
        for(int i=0;i<n;i++)total+=w[i]*marginal[i];
        double target=total/n;
        std::vector<double> next=w;
        for(int i=0;i<n;i++)
            next[i]*=std::sqrt(target/(w[i]*marginal[i]));
        double normalizer=std::accumulate(next.begin(),next.end(),0.0);
        for(double& v:next)v/=normalizer;
        double change=0;
        for(int i=0;i<n;i++)
            change=std::max(change,std::fabs(next[i]-w[i]));
        w=next;
        if(change<tolerance)break;
    }
    return w;
}

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>

namespace kline {
inline std::vector<double> drifted_portfolio_weights(
    const std::vector<double>& initial_weights,
    const std::vector<double>& period_returns,double cash_weight=0){
    check_finite(initial_weights);check_finite(period_returns);
    require(initial_weights.size()==period_returns.size()&&
            std::isfinite(cash_weight)&&cash_weight>=0,
            "Invalid portfolio drift weights");
    double assets=cash_weight;
    std::vector<double> next;
    for(int i=0;i<int(initial_weights.size());i++){
        require(initial_weights[i]>=0&&period_returns[i]>-1,
                "Long only drift requires nonnegative weights and viable returns");
        next.push_back(initial_weights[i]*(1+period_returns[i]));
        assets+=next.back();
    }
    require(assets>0,"Drifted portfolio has no value");
    for(double& v:next)v/=assets;
    return next;
}
inline std::vector<double> no_trade_band_rebalance(
    const std::vector<double>& current,
    const std::vector<double>& target,double maximum_deviation){
    check_finite(current);check_finite(target);
    require(current.size()==target.size()&&
            std::isfinite(maximum_deviation)&&maximum_deviation>=0,
            "Invalid no trade band portfolio");
    for(int i=0;i<int(current.size());i++)
        require(current[i]>=0&&target[i]>=0,
                "Rebalancing rule uses nonnegative long-only weights");
    bool needed=false;
    for(int i=0;i<int(current.size());i++)
        needed|=std::fabs(target[i]-current[i])>maximum_deviation;
    return needed?target:current;
}
