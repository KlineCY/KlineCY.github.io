namespace kline {
struct SeriesQuality {
    int observations=0;
    int missing=0;
    int valid=0;
    int repeated_values=0;
    double coverage=0;
};

inline SeriesQuality inspect_series(const std::vector<double>& values) {
    require(!values.empty(),"Series quality needs observations");
    SeriesQuality result;
    result.observations=int(values.size());
    for(int i=0;i<int(values.size());i++){
        if(!std::isfinite(values[i]))result.missing++;
        else {
            result.valid++;
            if(i>0 && std::isfinite(values[i-1])&&
               values[i]==values[i-1])result.repeated_values++;
        }
    }
    result.coverage=double(result.valid)/result.observations;
    return result;
}

inline std::vector<double> forward_fill(
    const std::vector<double>& values,int maximum_gap) {
    require(!values.empty()&&maximum_gap>=0,
            "Invalid forward-fill gap");
    std::vector<double> out=values;
    double last=0;
    bool known=false;
    int gap=0;
    for(double& value:out){
        if(std::isfinite(value)){
            known=true;last=value;gap=0;
        }else{
            gap++;
            if(known && gap<=maximum_gap)value=last;
        }
    }
    return out;
}

inline std::vector<int> available_rows(const std::vector<double>& values) {
    require(!values.empty(),"No observations to inspect");
    std::vector<int> indices;
    for(int i=0;i<int(values.size());i++)
        if(std::isfinite(values[i]))indices.push_back(i);
    return indices;
}

inline std::vector<double> lagged_signal(
    const std::vector<double>& values,int delay) {
    check_finite(values);
    require(delay>=1&&delay<int(values.size()),
            "Signal delay must be positive and shorter than history");
    std::vector<double> output;
    for(int i=delay;i<int(values.size());i++)
        output.push_back(values[i-delay]);
    return output;
}

inline double max_relative_jump(const std::vector<double>& prices) {
    check_finite(prices);
    require(prices.size()>=2,"Jump check needs prices");
    double largest=0;
    for(int i=1;i<int(prices.size());i++){
        require(prices[i]>0&&prices[i-1]>0,"Jump check needs positive prices");
        largest=std::max(largest,std::fabs(prices[i]/prices[i-1]-1));
    }
    return largest;
}

inline double stock_total_return(double old_price,double new_price,
                                  double cash_dividend,double split_ratio=1){
    require(old_price>0&&new_price>0&&split_ratio>0&&
            std::isfinite(old_price)&&std::isfinite(new_price)&&
            std::isfinite(cash_dividend)&&cash_dividend>=0&&
            std::isfinite(split_ratio),"Invalid corporate action");
    return (new_price*split_ratio+cash_dividend)/old_price-1;
}
inline std::vector<double> stock_total_returns(
    const std::vector<double>& close,
    const std::vector<double>& dividend,
    const std::vector<double>& split){
    check_finite(close);check_finite(dividend);check_finite(split);
    require(close.size()==dividend.size()&&close.size()==split.size()&&
            close.size()>=2,"Corporate action series shapes");
    std::vector<double> results;
    for(int t=1;t<int(close.size());t++)
        results.push_back(stock_total_return(close[t-1],close[t],
                                             dividend[t],split[t]));
    return results;
}
inline double rights_adjusted_theoretical_ex_price(double price_before,

struct TimeFold {
    int train_begin = 0;
    int train_end = 0; // Exclusive
    int test_begin = 0;
    int test_end = 0; // Exclusive
};

inline std::vector<TimeFold> walk_forward_folds(
    int observations, int training_window, int testing_window,
    int embargo = 0, bool expanding = false) {
    require(observations >= 1 && training_window >= 2 && testing_window >= 1
            && embargo >= 0, "Invalid walk-forward window parameters");
    std::vector<TimeFold> result;
    for (int train_end = training_window;
         train_end + embargo + testing_window <= observations;
         train_end += testing_window) {
        int first = expanding ? 0 : train_end - training_window;
        result.push_back({first, train_end, train_end + embargo,
                          train_end + embargo + testing_window});
    }
    return result;
}

inline std::vector<int> moving_block_resample(int observations, int block_length,
                                               unsigned int seed) {
    require(observations >= 2 && block_length > 0 &&
            block_length <= observations, "Invalid bootstrap block parameters");
    std::mt19937 random(seed);
    std::uniform_int_distribution<int> start(0, observations - block_length);
    std::vector<int> indices;
    while (int(indices.size()) < observations) {
        int beginning = start(random);
        for (int offset = 0; offset < block_length &&
             int(indices.size()) < observations; offset++) {
            indices.push_back(beginning + offset);
        }
    }
    return indices;
}

inline std::vector<double> block_bootstrap_means(const std::vector<double>& returns,
                                                   int block_length,
                                                   int replicates, unsigned int seed) {
    check_finite(returns);
    require(replicates > 0 && replicates <= 100000, "Invalid bootstrap replicates");
    std::vector<double> result;
    for (int r = 0; r < replicates; r++) {
        auto indices = moving_block_resample(int(returns.size()), block_length, seed + r);
        double sum = 0;
        for (int id : indices) sum += returns[id];
        result.push_back(sum / indices.size());
    }
    return result;
}

inline double bonferroni_adjusted_p(double raw_p, int number_of_tests) {
    require(raw_p >= 0 && raw_p <= 1 && number_of_tests > 0,
            "Invalid p-value or test count");
    return std::min(1.0, raw_p * number_of_tests);
}

inline std::vector<bool> benjamini_hochberg(const std::vector<double>& p_values,
                                             double false_discovery_rate = 0.05) {
    check_finite(p_values);
    require(false_discovery_rate > 0 && false_discovery_rate < 1,
            "Invalid target FDR");
    int m = int(p_values.size());
    std::vector<int> order(m);
    std::iota(order.begin(), order.end(), 0);
    for (double p : p_values) require(p >= 0 && p <= 1, "Invalid individual p-value");
    std::stable_sort(order.begin(), order.end(),
                     [&](int a, int b) { return p_values[a] < p_values[b]; });
    int k = 0;
    for (int i = 0; i < m; i++)
        if (p_values[order[i]] <= false_discovery_rate * (i + 1) / m) k = i + 1;
    std::vector<bool> rejected(m, false);
    for (int i = 0; i < k; i++) rejected[order[i]] = true;
    return rejected;
}

inline double brier_score(const std::vector<double>& probability,
                          const std::vector<int>& binary_label) {
    check_finite(probability);
    require(binary_label.size() == probability.size(), "Labels must align with predictions");
    double result = 0;
    for (int i = 0; i < int(probability.size()); i++) {
        require(probability[i] >= 0 && probability[i] <= 1 &&
                (binary_label[i] == 0 || binary_label[i] == 1),
                "Brier requires probabilities and binary labels");
        result += std::pow(probability[i] - binary_label[i], 2);
    }
    return result / probability.size();
}

struct CalibrationBin {
    int observations = 0;
    double predicted_frequency = 0;
    double observed_frequency = 0;
};

inline std::vector<CalibrationBin> reliability_bins(
    const std::vector<double>& probabilities, const std::vector<int>& labels,
    int bins = 10) {
    check_finite(probabilities);
    require(probabilities.size() == labels.size() && bins >= 1 && bins <= 100,
            "Invalid calibration bin inputs");
    std::vector<CalibrationBin> results(bins);
    for (int i = 0; i < int(probabilities.size()); i++) {
        require(probabilities[i] >= 0 && probabilities[i] <= 1 &&
                (labels[i] == 0 || labels[i] == 1), "Invalid probability or label");
        int index = std::min(bins - 1, int(probabilities[i] * bins));
        results[index].observations++;
        results[index].predicted_frequency += probabilities[i];
        results[index].observed_frequency += labels[i];
    }
    for (auto& bin : results) {
        if (bin.observations > 0) {
            bin.predicted_frequency /= bin.observations;
            bin.observed_frequency /= bin.observations;
        }
    }
    return results;
}

inline OLSResult entity_fixed_effects(
    const Matrix& features,const std::vector<double>& outcomes,
    const std::vector<int>& entity_ids){
    check_feature_table(features,outcomes);
    require(entity_ids.size()==outcomes.size(),
            "Entity panel identities differ");
    int n=int(features.size()),p=int(features.front().size());
    std::map<int,std::vector<int>> groups;
    for(int i=0;i<n;i++)groups[entity_ids[i]].push_back(i);
    require(int(groups.size())>=2,"Panel requires multiple entities");
    Matrix x=features;
    std::vector<double> y=outcomes;
    for(const auto& group:groups){
        const auto& ids=group.second;
        require(ids.size()>=2,"Each entity needs repeated observations");
        double ym=0;std::vector<double> xm(p,0);
        for(int index:ids){
            ym+=outcomes[index]/ids.size();
            for(int j=0;j<p;j++)xm[j]+=features[index][j]/ids.size();
        }
        for(int index:ids){
            y[index]-=ym;
            for(int j=0;j<p;j++)x[index][j]-=xm[j];
        }
    }
    auto result=ordinary_least_squares(x,y);
    return result;
}

inline double gpd_survival(double excess,GeneralizedPareto p){
    require(std::isfinite(excess)&&excess>=0&&
            std::isfinite(p.shape)&&std::isfinite(p.scale)&&p.scale>0,
            "GPD parameters");

    double base=1+p.shape*excess/p.scale;

inline double gpd_quantile(double probability,GeneralizedPareto p){
    require(std::isfinite(probability)&&probability>=0&&probability<1&&

            "GPD quantile parameters");

inline double hill_tail_index(std::vector<double> losses,int upper_count){

inline Matrix eigenvalue_floor_covariance(const Matrix& covariance,
                                           double fraction=1e-5){
    require(std::isfinite(fraction)&&fraction>=0,"Invalid spectral floor");
    auto eigen=symmetric_eigen(covariance);
    double largest=std::max(0.0,eigen.values.front());
    double floor=largest*fraction;
    int n=int(covariance.size());
    Matrix result(n,std::vector<double>(n,0));
    for(int k=0;k<n;k++){
        double value=std::max(floor,eigen.values[k]);
        for(int i=0;i<n;i++)
            for(int j=0;j<n;j++)
                result[i][j]+=value*eigen.vectors[i][k]*eigen.vectors[j][k];
    }
    return result;
}
inline Matrix covariance_correlation(const Matrix& covariance){
    check_covariance(covariance);
    int n=int(covariance.size());
    Matrix corr(n,std::vector<double>(n,0));
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++){
            double denom=std::sqrt(covariance[i][i]*covariance[j][j]);
            corr[i][j]=denom>0?covariance[i][j]/denom:0;
            if(i==j&&denom>0)corr[i][i]=1;
        }
    return corr;
}

#include <cmath>

namespace kline {
struct VaRBacktest{
    int observations=0,exceptions=0;
    double expected_exception_rate=0,realized_exception_rate=0;
    double kupiec_lr=0;
};
inline VaRBacktest kupiec_var_coverage(
    const std::vector<double>& realized_losses,
    const std::vector<double>& var_forecasts,double confidence){
    check_finite(realized_losses);check_finite(var_forecasts);
    require(realized_losses.size()==var_forecasts.size()&&
            std::isfinite(confidence)&&confidence>0&&confidence<1,
            "Invalid VaR coverage inputs");
    VaRBacktest r;
    r.observations=int(realized_losses.size());
    r.expected_exception_rate=1-confidence;
    for(int i=0;i<r.observations;i++)
        if(realized_losses[i]>var_forecasts[i])r.exceptions++;
    r.realized_exception_rate=double(r.exceptions)/r.observations;
    double p=r.expected_exception_rate,q=r.realized_exception_rate;
    double null_ll=r.exceptions*std::log(p)+
        (r.observations-r.exceptions)*std::log1p(-p);
    double alt_ll=0;
    if(r.exceptions>0)alt_ll+=r.exceptions*std::log(q);
    if(r.exceptions<r.observations)
        alt_ll+=(r.observations-r.exceptions)*std::log1p(-q);
    r.kupiec_lr=std::max(0.0,2*(alt_ll-null_ll));
    return r;
}
inline double expected_shortfall_residual(
    const std::vector<double>& realized_losses,
    const std::vector<double>& var_forecasts,
    const std::vector<double>& es_forecasts){
    check_finite(realized_losses);check_finite(var_forecasts);
    check_finite(es_forecasts);
    require(realized_losses.size()==var_forecasts.size()&&
            realized_losses.size()==es_forecasts.size(),
            "ES residual series shape");
    double sum=0;int count=0;
    for(int i=0;i<int(realized_losses.size());i++)
        if(realized_losses[i]>var_forecasts[i]){
            sum+=realized_losses[i]-es_forecasts[i];count++;
        }
    require(count>0,"No VaR exceptions for ES residual");
    return sum/count;
}
