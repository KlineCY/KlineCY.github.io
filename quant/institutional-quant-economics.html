#include "stats.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
inline double binary_kelly_fraction(double probability_win,
                                    double gain_on_win,double loss_on_lose){
    require(std::isfinite(probability_win)&&probability_win>=0&&
            probability_win<=1&&std::isfinite(gain_on_win)&&gain_on_win>0&&
            std::isfinite(loss_on_lose)&&loss_on_lose>0,
            "Invalid Kelly binary contract");
    double unconstrained=probability_win/loss_on_lose-
                         (1-probability_win)/gain_on_win;
    return std::max(0.0,std::min(1/loss_on_lose,unconstrained));
}
inline double fractional_kelly_fraction(double probability_win,
                double gain_on_win,double loss_on_lose,double shrink_fraction){
    require(std::isfinite(shrink_fraction)&&shrink_fraction>=0&&
            shrink_fraction<=1,"Fractional Kelly reduction outside [0,1]");
    return shrink_fraction*binary_kelly_fraction(
        probability_win,gain_on_win,loss_on_lose);
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
inline double herfindahl_concentration(const std::vector<double>& weights){
    check_finite(weights);
    double sum=0,hhi=0;
    for(double w:weights){
        require(w>=0,"Concentration requires nonnegative shares");
        sum+=w;hhi+=w*w;
    }
    require(sum>0,"Concentration weight mass zero");
    return hhi/(sum*sum);
}
inline double effective_holdings_count(const std::vector<double>& weights){
    return 1/herfindahl_concentration(weights);
}
} // namespace kline

3.5　数值算法及完整源码

以标准C++17实现可调用算法，不依赖第三方数值库。由等额现金流贴现现值等于初始借款本金求出支付额；每一期先按期初余额计算利息，再把固定支付额的剩余部分冲抵本金。

#pragma once
#include "stats.hpp"
#include <cmath>
#include <vector>
namespace kline {
struct PaymentRecord{int period=0;double interest=0,principal=0,balance=0;};
inline std::vector<PaymentRecord> fixed_rate_amortization(
    double principal,double rate_per_period,int periods){
    require(std::isfinite(principal)&&principal>0&&
            std::isfinite(rate_per_period)&&rate_per_period>=0&&
            periods>=1&&periods<=10000,"Invalid fixed loan schedule");
    double payment=rate_per_period==0?principal/periods:
         principal*rate_per_period/
         (1-std::pow(1+rate_per_period,-periods));
    double balance=principal;
    std::vector<PaymentRecord> schedule;
    for(int i=1;i<=periods;i++){
        double interest=balance*rate_per_period;
        double principal_payment=std::min(balance,payment-interest);
        if(i==periods)principal_payment=balance;
        balance=std::max(0.0,balance-principal_payment);
        schedule.push_back({i,interest,principal_payment,balance});
    }
    return schedule;
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
inline double information_ratio_of_active_returns(
    const std::vector<double>& active_returns,int periods_per_year=252){
    check_finite(active_returns);
    require(active_returns.size()>=2&&periods_per_year>0,
            "Information ratio needs observations and frequency");
    double deviation=std::sqrt(variance(active_returns));
    require(deviation>0,"Information ratio requires nonzero active risk");
    return mean(active_returns)/deviation*std::sqrt(double(periods_per_year));
}
inline double upside_downside_capture(
   const std::vector<double>& fund,const std::vector<double>& benchmark,bool upside){
    check_finite(fund);check_finite(benchmark);
    require(fund.size()==benchmark.size(),"Capture input dates mismatch");
    double fund_sum=0,bench_sum=0;
    for(int i=0;i<int(fund.size());i++)if(upside?benchmark[i]>0:benchmark[i]<0){
        fund_sum+=fund[i];bench_sum+=benchmark[i];
    }
    require(std::fabs(bench_sum)>1e-12,"No matching market capture observations");
    return fund_sum/bench_sum;
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
#include <vector>
namespace kline {
struct DatedTaxLot{int acquired_day=0;double shares=0,cost_per_share=0;};
inline double unrealized_tax_lot_profit(
    const std::vector<DatedTaxLot>& lots,double current_price){
    require(std::isfinite(current_price)&&current_price>=0,
            "Invalid tax lot market price");
    double result=0;
    for(auto lot:lots){
        require(std::isfinite(lot.shares)&&lot.shares>=0&&
                std::isfinite(lot.cost_per_share)&&lot.cost_per_share>=0,
                "Invalid tax lot shares or basis");
        result+=lot.shares*(current_price-lot.cost_per_share);
    }
    return result;
}
inline double held_more_than_days(const std::vector<DatedTaxLot>& lots,
                                  int current_day,int threshold_days){
    require(threshold_days>=0,"Negative holding threshold");
    double shares=0;
    for(auto lot:lots){
        require(lot.acquired_day<=current_day&&lot.shares>=0,
                "Invalid tax lot acquisition date");
        if(current_day-lot.acquired_day>threshold_days)shares+=lot.shares;
    }
    return shares;
}
} // namespace kline
