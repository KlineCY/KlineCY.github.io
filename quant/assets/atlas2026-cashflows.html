1．连续复利现金流现值

$$PV=\sum_i C_ie^{-rt_i}$$

经济含义与理论结构

按每笔支付的距离当前年数和连续复利折现率，将未来净现金流逐笔折现并求和，得到在该贴现口径下的现金流现值。

数值案例与复算

两笔现金流分别在第1年和第2年支付10与20，折现率为零，现值为30。

$$\mathrm{Output}\approx 30$$

2．现金流的现值加权平均支付时间

$$D=\frac{\sum_i t_iC_ie^{-rt_i}}{\sum_i C_ie^{-rt_i}}$$

经济含义与理论结构

将非负现金流的现值作为权重，对所有支付时间求加权平均，得到由合同支付结构决定的平均资本回收时间。

5．贴现现金流的插值回收年限

$$T^*=t_{j-1}+(t_j-t_{j-1})\frac{-S_{j-1}}{PV_j}$$

经济含义与理论结构

沿时间轴累计折现现金流，若在某个相邻支付区间首次由负转非负，则以该区间内的现金流线性插值估计回本时间。

#pragma once
#include "atlas2026_common.hpp"
namespace kline { namespace atlas2026 {
inline double cashflow_present_value(const Values& x,const Values& y,double p){valid(x,y,p);double sum=0;for(std::size_t i=0;i<x.size();++i){require(x[i]>=0&&(i==0||x[i]>x[i-1]),"Cash-flow dates must increase");sum+=y[i]*std::exp(-p*x[i]);}require(std::isfinite(sum),"Cash-flow PV overflow");return sum;}
inline double cashflow_discounted_duration(const Values& x,const Values& y,double p){valid(x,y,p);double sum=0,weighted=0;for(std::size_t i=0;i<x.size();++i){require(x[i]>=0&&y[i]>=0&&(i==0||x[i]>x[i-1]),"Invalid positive cash-flow schedule");double pv=y[i]*std::exp(-p*x[i]);sum+=pv;weighted+=x[i]*pv;}require(sum>0,"Positive cash-flow PV required");return weighted/sum;}
inline double cashflow_time_dispersion(const Values& x,const Values& y,double p){valid(x,y,p);double sum=0,a=0,b=0;for(std::size_t i=0;i<x.size();++i){require(x[i]>=0&&y[i]>=0&&(i==0||x[i]>x[i-1]),"Invalid time dispersion schedule");double v=y[i]*std::exp(-p*x[i]);sum+=v;a+=x[i]*v;b+=x[i]*x[i]*v;}require(sum>0,"Positive PV required");double d=a/sum;return std::sqrt(std::max(0.0,b/sum-d*d));}
inline double cashflow_coverage_ratio(const Values& x,const Values& y,double p){valid(x,y,p);double inc=0,out=0;for(std::size_t i=0;i<x.size();++i){require(x[i]>=0&&(i==0||x[i]>x[i-1]),"Invalid payment schedule");double v=y[i]*std::exp(-p*x[i]);inc+=std::max(0.0,v);out+=std::max(0.0,-v);}require(out>0,"Negative funding cashflow required");return inc/out;}
inline double cashflow_discounted_payback(const Values& x,const Values& y,double p){valid(x,y,p);require(y[0]<0&&x[0]>=0,"First cash flow must be an initial outflow");double accum=y[0]*std::exp(-p*x[0]);for(std::size_t i=1;i<x.size();++i){require(x[i]>x[i-1],"Cash-flow times must strictly increase");double change=y[i]*std::exp(-p*x[i]);double prev=accum;accum+=change;if(prev<0&&accum>=0){require(change>0,"Payback crossing requires inflow");return x[i-1]+(x[i]-x[i-1])*(-prev/change);}}throw std::invalid_argument("Discounted cash flows never recover initial outlay");}
} }
