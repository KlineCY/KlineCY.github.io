Black–Scholes有限差分：Crank–Nicolson、边界条件与三对角求解

系统化研究与实验　完整源码

经济机制与模型设定

欧式期权PDE以到期支付为初值，向剩余期限方向推进。均匀现货网格将扩散、漂移及折现转换为三对角空间算子，Crank–Nicolson对时间作梯形离散。上界是有限域近似，不能把网格截断误差与时间离散误差混淆。

推导与计算方法

每步使用旧时刻和新时刻边界值构造内部节点右端，Thomas算法求解三对角系统，并用未消元系数回代残差。末端按相邻现货节点线性插值。实现支持欧式看涨及看跌，零波动走确定性风险中性支付。

$$\partial_\tau V=\tfrac12\sigma^2S^2V_{SS}+(r-q)SV_S-rV$$
$$(L_hV)_i=a_iV_{i-1}+b_iV_i+c_iV_{i+1}$$
$$(I-\Delta\tau L_h/2)V^{k+1}=(I+\Delta\tau L_h/2)V^k$$
$$V_{\rm call}(0,\tau)=0,\quad V_{\rm call}(S_{\max},\tau)\approx S_{\max}e^{-q\tau}-Ke^{-r\tau}$$

可复算样本与解析基准

S=K=100、零波动、零利率和股息时价格0，网格20份、上界200、40步，空间步长10、时间步长0.025。非零波动通过Black–Scholes闭式、网格加密及看涨看跌平价分别检查。

研究边界及计算复杂度

O(MN)时间及O(M)存储，要求S在网格内、上界高于K。中央差分在对流占优时可能出现振荡；输出最低节点值用于诊断而非静默裁剪。真实美式期权需另行加入自由边界。

inline Report tridiagonal_cn_option(const Mat& x,const Vec& p){
    validate(x,1,6,6);parameters(p,4);auto M=count(p[0],4,2000),N=count(p[1],1,100000);double top=positive(p[2]);require(p[3]==0||p[3]==1,"Call flag must be binary");bool call=p[3];require(double(M)*N*x.size()<=1e8,"PDE work limit");Report o{{"Price","GridSpacing","TimeStep","MinimumGridValue","LinearResidual"},{}};for(auto&r:x){double S=positive(r[0]),K=positive(r[1]),rate=r[2],q=r[3],vol=r[4],T=positive(r[5]),ds=top/M,dt=T/N;require(S<top&&K<top&&vol>=0,"Invalid PDE domain");if(vol==0){double f=S*std::exp(-q*T)-K*std::exp(-rate*T);o.rows.push_back({std::max(call?f:-f,0.),ds,dt,0,0});continue;}auto bound=[&](double t){double v=top*std::exp(-q*t)-K*std::exp(-rate*t);return Vec{call?0:K*std::exp(-rate*t),call?std::max(0.,v):std::max(0.,-v)};};Vec V(M+1),a(M+1),b(M+1),c(M+1);for(std::size_t i=0;i<=M;++i){double z=i*ds-K;V[i]=std::max(call?z:-z,0.);if(i&&i<M){double v=vol*vol*i*i,w=(rate-q)*i;a[i]=.5*(v-w);b[i]=-v-rate;c[i]=.5*(v+w);}}double err=0;for(std::size_t t=0;t<N;++t){Vec next(M+1),l(M-1),d(M-1),u(M-1),rhs(M-1);auto bd=bound((t+1)*dt);next[0]=bd[0];next[M]=bd[1];for(std::size_t i=1;i<M;++i){auto j=i-1;l[j]=-.5*dt*a[i];d[j]=1-.5*dt*b[i];u[j]=-.5*dt*c[i];rhs[j]=V[i]+.5*dt*(a[i]*V[i-1]+b[i]*V[i]+c[i]*V[i+1]);}rhs[0]-=l[0]*next[0];rhs.back()-=u.back()*next[M];Vec diag=d,z=rhs;for(std::size_t j=1;j<M-1;++j){require(std::abs(diag[j-1])>1e-14,"Singular PDE pivot");double f=l[j]/diag[j-1];diag[j]-=f*u[j-1];z[j]-=f*z[j-1];}for(std::size_t j=M-1;j-->0;){require(std::abs(diag[j])>1e-14,"Singular PDE pivot");next[j+1]=(z[j]-(j+1<M-1?u[j]*next[j+2]:0))/diag[j];}for(std::size_t j=0;j<M-1;++j){double v=d[j]*next[j+1]+(j?l[j]*next[j]:0)+(j+1<M-1?u[j]*next[j+2]:0);err=std::max(err,std::abs(v-rhs[j]));}V=next;}auto i=std::min(M-1,std::size_t(S/ds));double w=S/ds-i;double price=(1-w)*V[i]+w*V[i+1];o.rows.push_back({price,ds,dt,*std::min_element(V.begin(),V.end()),err});}return o;
}
