2.5　数值计算步骤与源码

针对CRR二叉树下美式期权定价的核心数值步骤以C++17标准库编写。在每个时间步令标的价格乘以上涨因子u或下跌因子d，并由风险中性远期增长率求出p。终点使用到期收益，再逐层比较即时行权价值和下一层期望贴现价值。

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
#include <vector>
namespace kline {
inline double american_crr_option(double spot,double strike,double rate,
         double dividend_yield,double volatility,double maturity,
         int steps,bool call){
    require(std::isfinite(spot)&&spot>0&&std::isfinite(strike)&&strike>0&&
            std::isfinite(rate)&&std::isfinite(dividend_yield)&&
            std::isfinite(volatility)&&volatility>0&&
            std::isfinite(maturity)&&maturity>0&&steps>=1&&steps<=20000,
            "Invalid CRR option inputs");
    double dt=maturity/steps,up=std::exp(volatility*std::sqrt(dt)),
           down=1/up,p=(std::exp((rate-dividend_yield)*dt)-down)/
                         (up-down);
    require(p>=0&&p<=1,"CRR steps violate risk neutral probability bounds");
    double discount=std::exp(-rate*dt);
    std::vector<double> values(steps+1);
    for(int j=0;j<=steps;j++){
        double price=spot*std::pow(up,j)*std::pow(down,steps-j);
        values[j]=std::max(0.0,call?price-strike:strike-price);
    }
    for(int t=steps-1;t>=0;t--){
        for(int j=0;j<=t;j++){
            double price=spot*std::pow(up,j)*std::pow(down,t-j);
            double exercise=std::max(0.0,call?price-strike:strike-price);
            double hold=discount*(p*values[j+1]+(1-p)*values[j]);
            values[j]=std::max(exercise,hold);
        }
    }
    return values[0];
}
} // namespace kline

#include <cmath>
namespace kline {
struct Black76Greeks{double value=0,forward_delta=0,forward_gamma=0,vega=0;};
inline Black76Greeks black76_futures_option(
    double forward,double strike,double rate,double volatility,
    double maturity,bool call){
    require(std::isfinite(forward)&&forward>0&&std::isfinite(strike)&&
            strike>0&&std::isfinite(rate)&&std::isfinite(volatility)&&
            volatility>0&&std::isfinite(maturity)&&maturity>0,
            "Invalid Black 76 futures option");
    double standard=volatility*std::sqrt(maturity);
    double d1=(std::log(forward/strike)+0.5*standard*standard)/standard;
    double d2=d1-standard,df=std::exp(-rate*maturity);
    double call_value=df*(forward*standard_normal_cdf(d1)-
                          strike*standard_normal_cdf(d2));
    double put_value=call_value-df*(forward-strike);
    double delta=df*(call?standard_normal_cdf(d1):
                           standard_normal_cdf(d1)-1);
    return {call?call_value:put_value,delta,
            df*standard_normal_pdf(d1)/(forward*standard),
            df*forward*standard_normal_pdf(d1)*std::sqrt(maturity)};
}

inline BachelierPrice bachelier_option(
     double forward,double strike,double rate,double normal_vol,
     double maturity,bool call){
    require(std::isfinite(forward)&&std::isfinite(strike)&&
            std::isfinite(rate)&&std::isfinite(normal_vol)&&normal_vol>0&&

5.5　数值计算步骤与源码

针对Brownian Bridge障碍穿越概率的核心数值步骤以C++17标准库编写。对对数价格的扩散过程，条件在两个端点上的Brownian Bridge具有可计算的穿越分布。端点与障碍的对数距离进入指数，得到区间内未触碰概率。

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
inline double log_brownian_bridge_down_survival(
    double start_spot,double end_spot,double barrier,
    double volatility,double interval_years){
    require(std::isfinite(start_spot)&&std::isfinite(end_spot)&&
            std::isfinite(barrier)&&start_spot>0&&end_spot>0&&barrier>0&&
            std::isfinite(volatility)&&volatility>0&&
            std::isfinite(interval_years)&&interval_years>0,
            "Invalid Brownian bridge conditional survival");
    if(start_spot<=barrier||end_spot<=barrier)return 0;
    double a=std::log(start_spot/barrier),b=std::log(end_spot/barrier);
    return -std::expm1(-2*a*b/(volatility*volatility*interval_years));
}
inline double log_brownian_bridge_up_survival(
    double start_spot,double end_spot,double barrier,
    double volatility,double interval_years){
    require(std::isfinite(start_spot)&&std::isfinite(end_spot)&&
            std::isfinite(barrier)&&start_spot>0&&end_spot>0&&barrier>0&&
            std::isfinite(volatility)&&volatility>0&&
            std::isfinite(interval_years)&&interval_years>0,
            "Invalid up barrier bridge input");
    if(start_spot>=barrier||end_spot>=barrier)return 0;
    double a=std::log(barrier/start_spot),b=std::log(barrier/end_spot);
    return -std::expm1(-2*a*b/(volatility*volatility*interval_years));
}

6.5　数值计算步骤与源码

针对Black–Scholes隐含波动率反演的核心数值步骤以C++17标准库编写。先根据股息贴现现值、执行价贴现额确定期权价格的严格内在边界与最大边界，再利用期权对波动率的单调性做区间二分。

#pragma once
#include "derivatives.hpp"
#include <cmath>
namespace kline {
inline double implied_black_scholes_volatility(
    double option_price,double spot,double strike,double rate,
    double dividend_yield,double maturity,bool call){
    require(std::isfinite(option_price)&&option_price>=0&&
            std::isfinite(spot)&&spot>0&&std::isfinite(strike)&&strike>0&&
            std::isfinite(rate)&&std::isfinite(dividend_yield)&&
            std::isfinite(maturity)&&maturity>0,"Invalid implied vol contract");
    double dfq=std::exp(-dividend_yield*maturity),dfr=std::exp(-rate*maturity);
    double intrinsic=call?std::max(spot*dfq-strike*dfr,0.0):
                          std::max(strike*dfr-spot*dfq,0.0);
    double upper=call?spot*dfq:strike*dfr;
    require(option_price>intrinsic+1e-12&&option_price<upper-1e-12,
            "Option quote outside strict implied-volatility bounds");
    double lo=1e-10,hi=0.5;
    for(int repeat=0;repeat<30;repeat++){
        if(black_scholes(spot,strike,rate,dividend_yield,hi,maturity,call).value
           >=option_price)break;
        hi*=2;
    }
    require(black_scholes(spot,strike,rate,dividend_yield,hi,maturity,call).value
             >=option_price,"Implied volatility upper search failed");
    for(int repeat=0;repeat<100;repeat++){
        double mid=(lo+hi)/2;
        double model=black_scholes(spot,strike,rate,dividend_yield,
                                   mid,maturity,call).value;
        if(model<option_price)lo=mid;
        else hi=mid;
    }
    return (hi+lo)/2;
}
} // namespace kline
