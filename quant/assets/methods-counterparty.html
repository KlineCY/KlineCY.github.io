期限边际违约概率与离散CVA

模型与推导

从存活概率下降量而非累积违约概率取得区间边际违约概率；独立暴露及违约的离散近似将贴现预期暴露和边际概率相乘。累积CVA为各期损失贡献和。

$$\Delta PD_i=S(t_{i-1})-S(t_i)$$
$$\mathrm{CVA}\approx\mathrm{LGD}\sum_i D_i EE_i\Delta PD_i$$

研究应用与成立条件

存活从1降至0.98再至0.95，边际违约概率2％、3％，给定暴露及贴现后累计2.544。需风险中性违约曲线和合适暴露条件；错误方向风险须额外联合建模。

双Bernoulli违约联合分布与相关

模型与推导

联合违约概率决定四个互斥事件质量：双违约、各单违约及均不违约。非负约束导出Fréchet边界；指标协方差等于联合概率减边际概率乘积。

$$\max(0,p_1+p_2-1)\le p_{12}\le\min(p_1,p_2)$$
$$\rho_D=(p_{12}-p_1p_2)/\sqrt{p_1(1-p_1)p_2(1-p_2)}$$

研究应用与成立条件

边际0.1、0.2、联合0.05，独立联合仅0.02，违约相关0.25。此相关是二元事件相关，不等于资产潜变量相关，二者不能直接混用。

信用损失分层与期望损失率

模型与推导

损失首先由低于attachment的次级资本吸收，进入区间后逐元承担，达到detachment后封顶。按情景概率加权得到期望分层损失，其损失率需除以该层名义宽度。

$$L_{A,D}(L)=\min((L-A)_+,D-A)$$
$$\mathrm{EL}_{A,D}=\sum_s p_sL_{A,D}(L_s),\quad \ell_{A,D}=\mathrm{EL}_{A,D}/(D-A)$$

研究应用与成立条件

损失0、10％、40％对应概率0.5、0.3、0.2，5％至25％层期望损失5.5％总体名义，层损失率27.5％。这只是单期损失分布，不是完整CDO现金流定价。

常强度下回收本金的违约债券

模型与推导

违约强度恒定时生存分布指数型，违约密度λe^{-λt}。零息信用债价格等于生存偿付腿加在违约时立即回收固定本金比例的现金流积分。使用expm1处理r＋λ接近零。

$$S(T)=e^{-\lambda T},\quad P_{\mathrm{surv}}=Ne^{-(r+\lambda)T}$$
$$P_{\mathrm{rec}}=NR\lambda\int_0^T e^{-(r+\lambda)t}dt$$

研究应用与成立条件

本金100、零利率、强度ln2、回收率40％，一年生存50％，价格70，两年55。该模型为违约时本金回收，不能用于回收市场价值或有息债而不调整。

#pragma once
#include "evidence_common.hpp"
namespace kline { namespace methods_counterparty {
using namespace evidence;
inline Report discrete_cva(const Mat& x,const Vec& p){
    validate(x,1,3,3);
    parameters(p,1);
    require(p[0]>=0&&p[0]<=1,"Invalid LGD");
    double prev=1,total=0;
    Report o{{"Period","MarginalPD","Contribution","CumulativeCVA"},{}};
    for(std::size_t i=0;
    i<x.size();
    ++i){auto&r=x[i];
    positive(r[0]);
    require(r[1]>=0&&r[2]>=0&&r[2]<=prev,"Invalid exposure or survival");
    double dp=prev-r[2],v=p[0]*r[0]*r[1]*dp;
    total+=v;
    prev=r[2];
    o.rows.push_back({double(i+1),dp,v,total});
    }return o;
    
}

inline Report joint_default(const Mat& x,const Vec& p){
    validate(x,1,3,3);
    parameters(p,0);
    Report o{{"Neither","Only1","Only2","Both","Correlation"},{}};
    for(auto&r:x){double a=r[0],b=r[1],c=r[2];
    require(a>0&&a<1&&b>0&&b<1&&c>=std::max(0.,a+b-1)&&c<=std::min(a,b),"Invalid joint probability");
    o.rows.push_back({1-a-b+c,a-c,b-c,c,(c-a*b)/std::sqrt(a*(1-a)*b*(1-b))});
    }return o;
    
}

inline Report tranche_loss(const Mat& x,const Vec& p){
    validate(x,1,2,2);
    parameters(p,2);
    double a=p[0],b=p[1];
    require(a>=0&&b>a&&b<=1,"Invalid attachment/detachment");
    double s=0,m=0,t=0;
    for(auto&r:x){require(r[0]>=0&&r[0]<=1&&r[1]>=0&&r[1]<=1,"Invalid loss or probability");
    s+=r[1];
    m+=r[0]*r[1];
    t+=std::min(std::max(r[0]-a,0.),b-a)*r[1];
    }require(std::abs(s-1)<1e-9,"Probability sum");
    return scalar({"ExpectedPortfolioLoss","ExpectedTrancheLoss","TrancheLossRate"},{m,t,t/(b-a)});
    
}

inline Report constant_hazard_bond(const Mat& x,const Vec& p){
    validate(x,1,1,1);
    parameters(p,4);
    double N=positive(p[0]),r=p[1],lambda=p[2],rec=p[3];
    require(lambda>=0&&rec>=0&&rec<=1,"Invalid hazard or recovery");
    double k=r+lambda;
    Report o{{"Horizon","Survival","SurvivalLeg","RecoveryLeg","Price"},{}};
    for(auto&a:x){double t=a[0];
    require(t>=0,"Negative tenor");
    double s=std::exp(-lambda*t),v=N*std::exp(-k*t),u=N*rec*lambda*(k==0?t:-std::expm1(-k*t)/k);
    o.rows.push_back({t,s,v,u,v+u});
    }return o;
    
}
