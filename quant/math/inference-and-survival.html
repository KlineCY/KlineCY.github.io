1．kendall_rank_dependence.hpp

\(\tau_b=\frac{C-D}{\sqrt{(C+D+T_X)(C+D+T_Y)}}\)

理论推导与可复算数值示例

Kendall的Tau-b从所有观测对出发，分别统计同序对\(C\)、逆序对\(D\)，以及仅在X或Y出现并列的观测对\(T_X,T_Y\)。归一化分母自动反映并列值造成的有效排序信息减少：

$$\tau_b=\frac{C-D}{\sqrt{(C+D+T_X)(C+D+T_Y)}}.$$

示例：\(X=(1,2,3,4)\)、\(Y=(1,3,2,4)\)。六个观测对中五对同序、一对逆序，且无并列：

$$\tau_b=\frac{5-1}{\sqrt{6\cdot6}}=\frac23\approx0.666667.$$

该指标适合比较非线性单调关联或两个因子排名的一致性。横截面包含大量相同分数时，Tau-b与忽略并列值的Tau-a会产生差异。参考：SciPy Kendall相关。

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
inline double kendall_tau_b(const std::vector<double>& x,const std::vector<double>& y){
    check_finite(x);check_finite(y);
    require(x.size()==y.size()&&x.size()>=2,
            "Kendall needs aligned paired observations");
    double concordant=0,discordant=0,ties_x=0,ties_y=0;
    for(int i=0;i<int(x.size());i++)
      for(int j=i+1;j<int(x.size());j++){
        double dx=x[i]-x[j],dy=y[i]-y[j];
        if(dx==0&&dy==0)continue;
        if(dx==0){ties_x++;continue;}
        if(dy==0){ties_y++;continue;}
        if(dx*dy>0)concordant++;
        else discordant++;
      }
    double denominator=std::sqrt((concordant+discordant+ties_x)*
                                 (concordant+discordant+ties_y));
    require(denominator>0,"Kendall undefined for constant ranks");
    return (concordant-discordant)/denominator;
}
} // namespace kline

#pragma once
#include "hac_mean_inference.hpp"
#include <cmath>
namespace kline {
struct LossComparison{double average_difference=0,standard_error=0,standardized=0;};
inline LossComparison squared_forecast_loss_comparison(
    const std::vector<double>& truth,
    const std::vector<double>& forecast_one,
    const std::vector<double>& forecast_two,int hac_lags){
    check_finite(truth);check_finite(forecast_one);check_finite(forecast_two);
    require(truth.size()==forecast_one.size()&&
            truth.size()==forecast_two.size(),
            "Forecast loss lengths must match");
    std::vector<double> differences;
    for(int i=0;i<int(truth.size());i++){
        double a=std::pow(truth[i]-forecast_one[i],2);
        double b=std::pow(truth[i]-forecast_two[i],2);
        differences.push_back(a-b);
    }
    auto report=bartlett_hac_mean(differences,hac_lags);
    return {report.estimate,report.standard_error,report.t_stat};
}
} // namespace kline

\(DD_t=1-\frac{V_t}{\max_{s\le t}V_s}\)

理论推导与可复算数值示例

对已扣除费用后的组合净值\(V_t\)，先构造截至该时点的历史高水位线\(H_t\)，再计算相对历史峰值的回撤：

$$H_t=\max_{0\le s\le t}V_s,\qquad DD_t=1-\frac{V_t}{H_t}.$$

示例：净值路径为\(100,105,98,101,108\)，运行高水位线为\(100,105,105,105,108\)。第三个观测点的回撤为：

$$DD_2=1-\frac{98}{105}\approx0.066667,\qquad DD_3=1-\frac{101}{105}\approx0.038095.$$

最大回撤为约6.667%。在净值重新超过前期高水位线之前，共有两个期末观测点处于水下状态。回撤幅度、持续期和恢复时间应分别表述，以免将风险路径压缩为单一指标。

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
struct DrawdownTiming{double largest_fraction=0;int longest_underwater=0;
    int peak_index=0,trough_index=0;};
inline DrawdownTiming drawdown_timing(const std::vector<double>& equity){
    check_finite(equity);
    double running_peak=equity[0];
    require(running_peak>0,"Initial equity must be positive");
    DrawdownTiming result;
    int last_peak=0,underwater=0;
    for(int i=1;i<int(equity.size());i++){
        require(equity[i]>0,"Equity path cannot be nonpositive");
        if(equity[i]>=running_peak){
            running_peak=equity[i];last_peak=i;underwater=0;
        }else{
            underwater++;
            double drawdown=1-equity[i]/running_peak;
            if(drawdown>result.largest_fraction){
                result.largest_fraction=drawdown;
                result.peak_index=last_peak;result.trough_index=i;
            }
            result.longest_underwater=std::max(result.longest_underwater,underwater);
        }
    }
    return result;
}
} // namespace kline

4．markov_transition_estimation.hpp

\(\widehat P_{ij}=\frac{N_{ij}+a}{\sum_kN_{ik}+Ka}\)

理论推导与可复算数值示例

离散状态马尔可夫转移估计从相邻时点的计数\(N_{ij}\)开始。加入每个转移目标相同的平滑伪计数\(a\)后，行内归一化得到有效概率矩阵：

$$\widehat P_{ij}=\frac{N_{ij}+a}{\sum_kN_{ik}+Ka}.$$

示例：设两个状态的转移计数行为\((8,2)\)与\((3,7)\)，并令\(a=1\)，则：

$$\widehat P=\begin{pmatrix}9/12&3/12\\4/12&8/12\end{pmatrix},\qquad\pi\widehat P=\pi,\quad \pi=(4/7,3/7).$$

#pragma once
#include "portfolio.hpp"
#include <cmath>
#include <vector>
namespace kline {
inline Matrix categorical_transition_probabilities(
    const std::vector<int>& states,int categories,double pseudocount=0){
    require(states.size()>=2&&categories>=2&&categories<=1000&&
            std::isfinite(pseudocount)&&pseudocount>=0,
            "Invalid Markov state transition data");
    Matrix transitions(categories,std::vector<double>(categories,pseudocount));
    for(int i=1;i<int(states.size());i++){
        require(states[i]>=0&&states[i]<categories&&
                states[i-1]>=0&&states[i-1]<categories,"State out of range");
        transitions[states[i-1]][states[i]]++;
    }
    for(auto& row:transitions){
        double total=0;
        for(double v:row)total+=v;
        require(total>0,"Unvisited Markov state requires pseudocount");
        for(double& v:row)v/=total;
    }
    return transitions;
}
} // namespace kline

5．kaplan_meier_survival.hpp

\(\widehat S(t)=\prod_{t_i\le t}(1-\frac{d_i}{n_i})\)

理论推导与可复算数值示例

Kaplan–Meier估计将每个事件时点的条件存活概率相乘。\(n_i\)是该事件发生前仍处于风险集合中的数量，\(d_i\)是该时点发生事件的数量：

$$\widehat S(t)=\prod_{t_i\le t}\left(1-\frac{d_i}{n_i}\right).$$

示例：最初五个对象都处于风险集合。在\(t_1\)时点发生一个事件，之后有一个对象被删失，故\(t_2\)事件前风险集合人数为三；在\(t_2\)再次发生一个事件：

$$\widehat S(t_1)=\frac45=0.8,\qquad \widehat S(t_2)=\frac45\cdot\frac23=\frac8{15}\approx0.533333.$$

删失对象离开后不再进入后续风险集合，但删失时点不应作为事件计入\(d_i\)。独立删失假设对于把经验曲线解释为总体存活概率具有关键意义。

5.5　数值步骤及完整C++17实现

在部分个体尚未观察到违约、退出或策略失败事件的情况下，使用右删失记录估计事件时间的生存分布。下面保留该算法的完整实现；代码包含输入检查和明确的数值处理方式，能够直接纳入已有C++17研究工具。

Kaplan–Meier删失生存曲线

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
#include <utility>
namespace kline {
struct SurvivalStep{double time=0,survival=1;int at_risk=0,defaults=0;};
inline std::vector<SurvivalStep> kaplan_meier(
    const std::vector<double>& durations,const std::vector<int>& event){
    check_finite(durations);
    require(durations.size()==event.size(),
            "Survival time and censor indicator lengths differ");
    std::vector<std::pair<double,int>> data;
    for(int i=0;i<int(durations.size());i++){
        require(durations[i]>=0&&(event[i]==0||event[i]==1),
                "Invalid survival observation");
        data.push_back({durations[i],event[i]});
    }
    std::sort(data.begin(),data.end());
    std::vector<SurvivalStep> result;
    double survival=1;
    int index=0,n=int(data.size());
    while(index<n){
        double time=data[index].first;
        int end=index,defaults=0;
        while(end<n&&data[end].first==time)defaults+=data[end++].second;
        int at_risk=n-index;
        survival*=1-double(defaults)/at_risk;
        result.push_back({time,survival,at_risk,defaults});
        index=end;
    }
    return result;
}
} // namespace kline
