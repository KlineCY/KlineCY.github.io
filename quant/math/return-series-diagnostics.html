#pragma once
#include "stats.hpp"
#include <cmath>
#include <vector>
namespace kline {
inline double overlapping_variance_ratio(const std::vector<double>& increments,int horizon){
    check_finite(increments);
    int n=int(increments.size());
    require(horizon>=2&&horizon<n/2,"Variance ratio horizon must be below half sample");
    double center=mean(increments),one_period=0;
    for(double v:increments)one_period+=(v-center)*(v-center);
    one_period/=n;
    require(one_period>0,"Constant increments have undefined variance ratio");
    double multi=0;
    for(int t=horizon-1;t<n;t++){
        double sum=0;
        for(int j=0;j<horizon;j++)sum+=increments[t-j]-center;
        multi+=sum*sum;
    }
    multi/=n-horizon+1;
    return multi/(horizon*one_period);
}
inline double overlapping_variance_ratio_deviation(const std::vector<double>& increments,int horizon){
    return overlapping_variance_ratio(increments,horizon)-1;
}
} // namespace kline

namespace kline {
inline double ljung_box_q_statistic(const std::vector<double>& residuals,int lags){
    check_finite(residuals);
    require(lags>=1&&lags<int(residuals.size())-1,"Insufficient Ljung Box lags");
    auto acf=autocorrelations(residuals,lags);
    double n=double(residuals.size()),q=0;
    for(int k=1;k<=lags;k++)q+=acf[k]*acf[k]/(n-k);
    return n*(n+2)*q;
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
struct NormalityMoments{double skewness=0,excess_kurtosis=0,jarque_bera=0;};
inline NormalityMoments normality_moments(const std::vector<double>& sample){
    check_finite(sample);
    require(sample.size()>=4,"Normality diagnostics need four observations");
    double mu=mean(sample),m2=0,m3=0,m4=0;
    for(double value:sample){
        double d=value-mu;
        m2+=d*d;m3+=d*d*d;m4+=d*d*d*d;
    }
    double n=double(sample.size());
    m2/=n;m3/=n;m4/=n;
    require(m2>0,"Normality moments undefined for constant series");
    NormalityMoments result;
    result.skewness=m3/std::pow(m2,1.5);
    result.excess_kurtosis=m4/(m2*m2)-3;
    result.jarque_bera=n*(result.skewness*result.skewness+
                        result.excess_kurtosis*result.excess_kurtosis/4)/6;
    return result;
}
} // namespace kline

inline void check_probability_vector(const std::vector<double>& p){
    check_finite(p);
    double sum=0;
    for(double value:p){
        require(value>=0,"Negative probability");
        sum+=value;
    }
    require(std::fabs(sum-1)<1e-9,"Probability masses must sum to one");
}
inline double discrete_entropy(const std::vector<double>& probabilities){
    check_probability_vector(probabilities);
    double result=0;
    for(double p:probabilities)if(p>0)result-=p*std::log(p);
    return result;
}
inline double kl_divergence(const std::vector<double>& p,const std::vector<double>& q){
    check_probability_vector(p);check_probability_vector(q);
    require(p.size()==q.size(),"KL distribution dimensions differ");
    double result=0;
    for(int i=0;i<int(p.size());i++){
        if(p[i]==0)continue;
        require(q[i]>0,"KL infinite due to zero reference mass");
        result+=p[i]*std::log(p[i]/q[i]);
    }
    return result;
}
inline double jensen_shannon_divergence(const std::vector<double>& p,
                                       const std::vector<double>& q){
    check_probability_vector(p);check_probability_vector(q);
    require(p.size()==q.size(),"JS distribution shape");
    std::vector<double> midpoint;
    for(int i=0;i<int(p.size());i++)midpoint.push_back((p[i]+q[i])/2);
    return (kl_divergence(p,midpoint)+kl_divergence(q,midpoint))/2;
}

#pragma once
#include "stats.hpp"
#include <algorithm>
#include <cmath>
namespace kline {
inline double lower_tail_conditional_exceedance(
   const std::vector<double>& x,const std::vector<double>& y,double tail){
    check_finite(x);check_finite(y);
    require(x.size()==y.size()&&x.size()>=8&&
            std::isfinite(tail)&&tail>0&&tail<0.5,
            "Invalid empirical lower tail dependence inputs");
    double cut_x=quantile(x,tail),cut_y=quantile(y,tail);
    int conditioning=0,joint=0;
    for(int i=0;i<int(x.size());i++){
        if(x[i]<=cut_x){
            conditioning++;
            if(y[i]<=cut_y)joint++;
        }
    }
    require(conditioning>0,"No tail conditioning observations");
    return double(joint)/conditioning;
}
inline double upper_tail_conditional_exceedance(
   const std::vector<double>& x,const std::vector<double>& y,double tail){
    check_finite(x);check_finite(y);
    require(x.size()==y.size()&&x.size()>=8&&
            std::isfinite(tail)&&tail>0&&tail<0.5,
            "Invalid empirical upper tail dependence inputs");
    double cut_x=quantile(x,1-tail),cut_y=quantile(y,1-tail);
    int conditioning=0,joint=0;
    for(int i=0;i<int(x.size());i++){
        if(x[i]>=cut_x){
            conditioning++;
            if(y[i]>=cut_y)joint++;
        }
    }
    require(conditioning>0,"No upper conditioning observations");
    return double(joint)/conditioning;
}
} // namespace kline

#pragma once
#include "stats.hpp"
#include <cmath>
namespace kline {
struct HACMean{double estimate=0,long_run_variance=0,standard_error=0,t_stat=0;};
inline HACMean bartlett_hac_mean(const std::vector<double>& data,int lags){
    check_finite(data);
    int n=int(data.size());
    require(n>=3&&lags>=0&&lags<n-1,
            "Invalid Newey West mean lag count");
    double mu=mean(data),cov0=0;
    for(double v:data)cov0+=(v-mu)*(v-mu)/n;
    double long_run=cov0;
    for(int lag=1;lag<=lags;lag++){
        double autocov=0;
        for(int t=lag;t<n;t++)autocov+=
           (data[t]-mu)*(data[t-lag]-mu)/n;
        long_run+=2*(1-double(lag)/(lags+1))*autocov;
    }
    require(long_run>=-1e-12,"Negative HAC long run covariance");
    long_run=std::max(0.0,long_run);
    double se=std::sqrt(long_run/n);
    return {mu,long_run,se,se>0?mu/se:0};
}
} // namespace kline
